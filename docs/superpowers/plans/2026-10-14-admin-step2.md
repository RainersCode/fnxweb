```ts
// Our photos are public URLs like <supabase>/storage/v1/object/public/media/<path>
export function storagePathFromUrl(url: string | null | undefined): string | null {
  if (!url) return null
  const prefix = `${process.env.NEXT_PUBLIC_SUPABASE_URL ?? ''}/storage/v1/object/public/media/`
  if (!url.startsWith(prefix)) return null
  try {
    return decodeURIComponent(url.slice(prefix.length).split('?')[0])
  } catch {
    return null
  }
}
```

```ts
import { createAdminClient } from './supabase'
import { storagePathFromUrl } from './storage-paths'

// Best effort: a failed file removal is logged and never undoes the database change
export async function removeMediaFiles(urls: Array<string | null | undefined>): Promise<void> {
  const paths = Array.from(new Set(urls.map(storagePathFromUrl).filter((p): p is string => Boolean(p))))
  if (!paths.length) return
  const db = createAdminClient()
  for (let i = 0; i < paths.length; i += 100) {
    const { error } = await db.storage.from('media').remove(paths.slice(i, i + 100))
    if (error) console.error('Error removing media files:', error)
  }
}
```

```ts
export type QueueStatus = 'waiting' | 'uploading' | 'done' | 'failed'
export type QueueItem = { id: string; name: string; status: QueueStatus; progress: number; error?: string; order: number }

const collator = new Intl.Collator('lv', { numeric: true, sensitivity: 'base' })

// Shooting order: DSC_9 before DSC_10
export function naturalSortByName<T extends { name: string }>(files: T[]): T[] {
  return [...files].sort((a, b) => collator.compare(a.name, b.name))
}

export function pickNextToStart(items: QueueItem[], limit = 3): string[] {
  const active = items.filter((i) => i.status === 'uploading').length
  const free = Math.max(0, limit - active)
  return items
    .filter((i) => i.status === 'waiting')
    .sort((a, b) => a.order - b.order)
    .slice(0, free)
    .map((i) => i.id)
}

export function queueSummary(items: QueueItem[]) {
  return {
    total: items.length,
    done: items.filter((i) => i.status === 'done').length,
    failed: items.filter((i) => i.status === 'failed').length,
    active: items.filter((i) => i.status === 'uploading').length,
    waiting: items.filter((i) => i.status === 'waiting').length,
  }
}

export function retryFailed(items: QueueItem[], ids?: string[]): QueueItem[] {
  return items.map((i) =>
    i.status === 'failed' && (!ids || ids.includes(i.id)) ? { ...i, status: 'waiting', progress: 0, error: undefined } : i
  )
}
```

```ts
export function galleryCover(coverUrl: string | null | undefined, firstImageUrl: string | null | undefined): string | null {
  return coverUrl || firstImageUrl || null
}

export function moveId(ids: string[], activeId: string, overId: string): string[] {
  const from = ids.indexOf(activeId)
  const to = ids.indexOf(overId)
  if (from < 0 || to < 0 || from === to) return ids
  const next = [...ids]
  const [moved] = next.splice(from, 1)
  next.splice(to, 0, moved)
  return next
}
```

```ts
export async function apiSend<T>(url: string, method: 'POST' | 'PATCH' | 'PUT' | 'DELETE', body?: unknown): Promise<T> {
  const res = await fetch(url, {
    method,
    headers: body !== undefined ? { 'content-type': 'application/json' } : undefined,
    body: body !== undefined ? JSON.stringify(body) : undefined,
  })
  const data = await res.json().catch(() => ({}))
```

```ts
import { NextRequest, NextResponse } from 'next/server'
import type { z } from 'zod'
import { isAdminRequest } from './admin-auth'
import { createAdminClient } from './supabase'
import { removeMediaFiles } from './storage'
import { fieldErrors } from './admin-schemas'

export type CrudOptions = {
  table: string
  schema: z.ZodTypeAny
  order: Array<{ column: string; ascending?: boolean; nullsFirst?: boolean }>
  imageField?: string
  label: string // Latvian noun for messages, e.g. "spēlētāju"
}

type Ctx = { params: { id: string } }

const unauthorized = () => NextResponse.json({ error: 'Unauthorized' }, { status: 401 })

async function parse(req: NextRequest, schema: z.ZodTypeAny) {
  const result = schema.safeParse(await req.json().catch(() => null))
  if (!result.success) {
    return { error: NextResponse.json({ error: 'Pārbaudi laukus', fields: fieldErrors(result.error) }, { status: 400 }) }
  }
  return { data: result.data as Record<string, unknown> }
}

// GET list + POST create for one table
export function listAndCreate(opts: CrudOptions) {
  return {
    GET: async () => {
      if (!(await isAdminRequest())) return unauthorized()
      let query = createAdminClient().from(opts.table).select('*')
      for (const o of opts.order) query = query.order(o.column, { ascending: o.ascending ?? true, nullsFirst: o.nullsFirst })
      const { data, error } = await query
      if (error) {
        console.error(`List ${opts.table}:`, error)
        return NextResponse.json({ error: 'Neizdevās ielādēt' }, { status: 500 })
      }
      return NextResponse.json(data)
    },
    POST: async (req: NextRequest) => {
      if (!(await isAdminRequest())) return unauthorized()
      const parsed = await parse(req, opts.schema)
      if (parsed.error) return parsed.error
      const { data, error } = await createAdminClient().from(opts.table).insert(parsed.data).select('*').single()
      if (error) {
        console.error(`Create ${opts.table}:`, error)
        return NextResponse.json({ error: `Neizdevās saglabāt ${opts.label}` }, { status: 500 })
      }
      return NextResponse.json(data)
    },
  }
}

// GET / PATCH / DELETE one row; replaced or removed photos are deleted after the database change
export function itemRoutes(opts: CrudOptions) {
  const load = (id: string) => createAdminClient().from(opts.table).select('*').eq('id', id).maybeSingle()
  return {
    GET: async (_req: NextRequest, { params }: Ctx) => {
      if (!(await isAdminRequest())) return unauthorized()
      const { data, error } = await load(params.id)
      if (error) return NextResponse.json({ error: 'Neizdevās ielādēt' }, { status: 500 })
      return data ? NextResponse.json(data) : NextResponse.json({ error: 'Nav atrasts' }, { status: 404 })
    },
    PATCH: async (req: NextRequest, { params }: Ctx) => {
      if (!(await isAdminRequest())) return unauthorized()
      const parsed = await parse(req, opts.schema)
      if (parsed.error) return parsed.error
      const { data: current } = await load(params.id)
      if (!current) return NextResponse.json({ error: 'Nav atrasts' }, { status: 404 })
      const { data, error } = await createAdminClient().from(opts.table).update(parsed.data).eq('id', params.id).select('*').single()
      if (error) {
        console.error(`Update ${opts.table}:`, error)
        return NextResponse.json({ error: `Neizdevās saglabāt ${opts.label}` }, { status: 500 })
      }
      if (opts.imageField) {
        const before = (current as Record<string, unknown>)[opts.imageField] as string | null
        const after = (data as Record<string, unknown>)[opts.imageField] as string | null
        if (before && before !== after) await removeMediaFiles([before])
      }
      return NextResponse.json(data)
    },
    DELETE: async (_req: NextRequest, { params }: Ctx) => {
      if (!(await isAdminRequest())) return unauthorized()
      const { data: current } = await load(params.id)
      const { error } = await createAdminClient().from(opts.table).delete().eq('id', params.id)
      if (error) {
        console.error(`Delete ${opts.table}:`, error)
        return NextResponse.json({ error: `Neizdevās dzēst ${opts.label}` }, { status: 500 })
      }
      if (opts.imageField && current) await removeMediaFiles([(current as Record<string, unknown>)[opts.imageField] as string | null])
      return NextResponse.json({ success: true })
    },
  }
}
```

```ts
import { useEffect } from 'react'

// Closing the tab or following a link with unsaved changes asks first
export function useUnsavedGuard(dirty: boolean) {
  useEffect(() => {
    if (!dirty) return
    const warn = (e: BeforeUnloadEvent) => { e.preventDefault(); e.returnValue = '' }
    const guardLinks = (e: MouseEvent) => {
      const link = (e.target as HTMLElement | null)?.closest('a[href]') as HTMLAnchorElement | null
      if (!link || link.target === '_blank') return
      if (!window.confirm('Ir nesaglabātas izmaiņas. Iziet bez saglabāšanas?')) {
        e.preventDefault()
        e.stopPropagation()
      }
    }
    window.addEventListener('beforeunload', warn)
    document.addEventListener('click', guardLinks, true)
    return () => {
      window.removeEventListener('beforeunload', warn)
      document.removeEventListener('click', guardLinks, true)
    }
  }, [dirty])
}
```

```tsx
'use client'

import { useState } from 'react'
import { useRouter } from 'next/navigation'
import { toast } from 'sonner'
import { Switch } from '@/components/ui/switch'
import { AdminFormPage } from '@/components/admin/ui/admin-form-page'
import { Field, inputClass } from '@/components/admin/ui/field'
import { PhotoField } from '@/components/admin/ui/photo-field'
import { apiSend } from '@/lib/admin-fetch'
import { coachSchema, fieldErrors, playerSchema } from '@/lib/admin-schemas'
import type { PersonKind, PersonRow } from './people-list'

const POSITIONS = ['Balsts', 'Āķētājs', 'Otrās rindas uzbrucējs', 'Flangs', 'Astotais', 'Saspēles vadītājs', 'Pussargs', 'Centrs', 'Malējais', 'Aizsargs']

const copy = {
  player: { api: '/api/admin/players', base: '/admin/players', back: 'Spēlētāji', newTitle: 'Jauns spēlētājs', del: 'Dzēst šo spēlētāju?', folder: 'players' },
  coach: { api: '/api/admin/coaches', base: '/admin/coaches', back: 'Treneri', newTitle: 'Jauns treneris', del: 'Dzēst šo treneri?', folder: 'coaches' },
}

type Form = { name: string; image_url: string | null; is_active: boolean; bio: string; jersey_number: string; position: string; role: string }

const fromRow = (r?: PersonRow & { bio?: string | null }): Form => ({
  name: r?.name ?? '', image_url: r?.image_url ?? null, is_active: r?.is_active ?? true, bio: r?.bio ?? '',
  jersey_number: r?.jersey_number != null ? String(r.jersey_number) : '', position: r?.position ?? '', role: r?.role ?? '',
})

export function PersonForm({ kind, initial }: { kind: PersonKind; initial?: PersonRow & { bio?: string | null } }) {
  const router = useRouter()
  const c = copy[kind]
  const [form, setForm] = useState<Form>(() => fromRow(initial))
  const [errors, setErrors] = useState<Record<string, string>>({})
  const [saving, setSaving] = useState(false)
  const [dirty, setDirty] = useState(false)

  const set = (patch: Partial<Form>) => { setForm((f) => ({ ...f, ...patch })); setDirty(true) }

  const save = async () => {
    const payload = kind === 'player'
      ? { name: form.name, image_url: form.image_url, is_active: form.is_active, bio: form.bio, position: form.position, jersey_number: form.jersey_number === '' ? null : Number(form.jersey_number) }
      : { name: form.name, image_url: form.image_url, is_active: form.is_active, bio: form.bio, role: form.role }
    const result = (kind === 'player' ? playerSchema : coachSchema).safeParse(payload)
    if (!result.success) { setErrors(fieldErrors(result.error)); return }
    setErrors({})
    setSaving(true)
    try {
      await apiSend(initial ? `${c.api}/${initial.id}` : c.api, initial ? 'PATCH' : 'POST', result.data)
      setDirty(false)
      toast.success('Saglabāts')
      router.push(c.base)
      router.refresh()
    } catch (error) {
      toast.error(error instanceof Error ? error.message : 'Neizdevās saglabāt')
    } finally {
      setSaving(false)
    }
  }

  const remove = async () => {
    if (!initial) return
    try {
      await apiSend(`${c.api}/${initial.id}`, 'DELETE')
      setDirty(false)
      toast.success('Dzēsts')
      router.push(c.base)
      router.refresh()
    } catch (error) {
      toast.error(error instanceof Error ? error.message : 'Neizdevās dzēst')
    }
  }

  return (
    <AdminFormPage backHref={c.base} backLabel={c.back} title={initial ? initial.name : c.newTitle} saving={saving} dirty={dirty} onSave={save} onDelete={initial ? remove : undefined} deleteTitle={c.del}>
      <PhotoField value={form.image_url} folder={c.folder} aspect="portrait" onChange={(url) => set({ image_url: url })} />
      <Field label="Vārds, uzvārds" htmlFor="name" error={errors.name}>
        <input id="name" value={form.name} onChange={(e) => set({ name: e.target.value })} className={inputClass} />
      </Field>
      {kind === 'player' ? (
        <div className="grid gap-5 sm:grid-cols-[120px_1fr]">
          <Field label="Numurs" htmlFor="num" error={errors.jersey_number}>
            <input id="num" inputMode="numeric" value={form.jersey_number} onChange={(e) => set({ jersey_number: e.target.value.replace(/\D/g, '').slice(0, 2) })} className={inputClass} />
          </Field>
          <Field label="Pozīcija" htmlFor="pos" error={errors.position}>
            <input id="pos" list="positions" value={form.position} onChange={(e) => set({ position: e.target.value })} className={inputClass} />
            <datalist id="positions">{POSITIONS.map((p) => <option key={p} value={p} />)}</datalist>
          </Field>
        </div>
      ) : (
        <Field label="Loma" htmlFor="role" error={errors.role}>
          <input id="role" value={form.role} onChange={(e) => set({ role: e.target.value })} placeholder="Galvenais treneris" className={inputClass} />
        </Field>
      )}
      <Field label="Par sevi" htmlFor="bio" error={errors.bio}>
        <textarea id="bio" rows={5} value={form.bio} onChange={(e) => set({ bio: e.target.value })} className={inputClass} />
      </Field>
      <label className="flex items-center justify-between gap-3 rounded-lg border border-[#DCE3DF] p-4">
        <span><span className="block font-semibold">Aktīvs</span><span className="text-sm text-[#5B6863]">Neaktīvie netiek rādīti mājaslapā</span></span>
        <Switch checked={form.is_active} onCheckedChange={(v) => set({ is_active: v })} />
      </label>
    </AdminFormPage>
  )
}
```

```tsx
import { fromDateTimeLocal, toDateTimeLocal } from '@/lib/articles'
import { findTeamByName, fixtureLogos, teamLogoFromFixture } from '@/lib/teams'
import { TeamPicker } from './team-picker'
import type { TeamRow } from './team-dialog'

export type FixtureRow = {
  id: string; opponent: string; match_date: string; location: string | null; score: string | null; is_home_game: boolean
  result: 'win' | 'loss' | 'draw' | null; description: string | null; home_logo_url: string | null; away_logo_url: string | null
}

const defaultDate = () => {
  const d = new Date(Date.now() + 7 * 24 * 3600 * 1000)
  d.setHours(14, 0, 0, 0)
  return toDateTimeLocal(d.toISOString())
}

export function FixtureForm({ initial, teams: initialTeams }: { initial?: FixtureRow; teams: TeamRow[] }) {
  const router = useRouter()
  const [teams, setTeams] = useState(initialTeams)
  const [team, setTeam] = useState<TeamRow | null>(() => findTeamByName(initialTeams, initial?.opponent) ?? null)
  const [form, setForm] = useState({
    is_home_game: initial?.is_home_game ?? true,
    when: initial ? toDateTimeLocal(initial.match_date) : defaultDate(),
    location: initial?.location ?? '',
    score: initial?.score ?? '',
    result: (initial?.result ?? '') as '' | 'win' | 'loss' | 'draw',
    description: initial?.description ?? '',
  })
  const [errors, setErrors] = useState<Record<string, string>>({})
  const [saving, setSaving] = useState(false)
  const [dirty, setDirty] = useState(false)
  const set = (patch: Partial<typeof form>) => { setForm((f) => ({ ...f, ...patch })); setDirty(true) }

  const played = form.when ? new Date(form.when) < new Date() : false

  const save = async () => {
    // Without a picked team an old match keeps its stored opponent and logo
    const opponent = team?.name ?? initial?.opponent ?? ''
    const logo = team ? team.logo_url : initial ? teamLogoFromFixture(initial) : null
    const payload = {
      opponent,
      match_date: form.when ? fromDateTimeLocal(form.when) : '',
      is_home_game: form.is_home_game,
      location: form.location,
      score: played ? form.score : '',
      result: played && form.result ? form.result : null,
      description: form.description,
      ...fixtureLogos(logo, form.is_home_game),
    }
    const result = fixtureSchema.safeParse(payload)
    if (!result.success) { setErrors(fieldErrors(result.error)); return }
    setErrors({})
    setSaving(true)
    try {
      await apiSend(initial ? `/api/admin/fixtures/${initial.id}` : '/api/admin/fixtures', initial ? 'PATCH' : 'POST', result.data)
      setDirty(false)
      toast.success('Saglabāts')
      router.push('/admin/fixtures')
      router.refresh()
    } catch (error) {
      toast.error(error instanceof Error ? error.message : 'Neizdevās saglabāt')
    } finally {
      setSaving(false)
    }
  }

  const remove = async () => {
    if (!initial) return
    try {
      await apiSend(`/api/admin/fixtures/${initial.id}`, 'DELETE')
      setDirty(false)
      toast.success('Dzēsts')
      router.push('/admin/fixtures')
      router.refresh()
    } catch (error) {
      toast.error(error instanceof Error ? error.message : 'Neizdevās dzēst')
    }
  }

  return (
    <AdminFormPage backHref="/admin/fixtures" backLabel="Spēles" title={initial ? `Fēnikss v ${initial.opponent}` : 'Jauna spēle'} saving={saving} dirty={dirty} onSave={save} onDelete={initial ? remove : undefined} deleteTitle="Dzēst šo spēli?">
      <Field label="Pretinieks" error={errors.opponent}>
        <TeamPicker teams={teams} value={team} fallbackName={initial?.opponent} onChange={(t) => { setTeam(t); setDirty(true) }} onTeamsChange={setTeams} />
      </Field>
      <div className="grid grid-cols-2 rounded-md bg-club-pale p-1" role="radiogroup" aria-label="Mājas vai izbraukumā">
```

```ts
'use client'

import { useCallback, useEffect, useRef, useState } from 'react'
import { naturalSortByName, pickNextToStart, queueSummary, retryFailed, type QueueItem } from '@/lib/upload-queue'

type Upload = (file: File, order: number, onProgress: (p: number) => void) => Promise<void>

// Uploads up to 3 files at once; failed files wait for "Mēģināt vēlreiz"
export function useUploadQueue(upload: Upload) {
  const [items, setItems] = useState<QueueItem[]>([])
  const files = useRef(new Map<string, File>())
  const previews = useRef(new Map<string, string>())
  const uploadRef = useRef(upload)
  uploadRef.current = upload

  const add = useCallback((list: File[], startOrder: number) => {
    const sorted = naturalSortByName(list.filter((f) => f.type.startsWith('image/') || /\.(heic|heif)$/i.test(f.name)))
    const created = sorted.map((file, i) => {
      const id = crypto.randomUUID()
      files.current.set(id, file)
      previews.current.set(id, URL.createObjectURL(file))
      return { id, name: file.name, status: 'waiting' as const, progress: 0, order: startOrder + i }
    })
    setItems((prev) => [...prev, ...created])
  }, [])

  useEffect(() => {
    const ids = pickNextToStart(items)
    if (!ids.length) return
    setItems((prev) => prev.map((i) => (ids.includes(i.id) ? { ...i, status: 'uploading', progress: 0 } : i)))
    for (const id of ids) {
      const file = files.current.get(id)
      const item = items.find((i) => i.id === id)
      if (!file || !item) continue
      uploadRef.current(file, item.order, (p) => setItems((prev) => prev.map((i) => (i.id === id ? { ...i, progress: p } : i))))
        .then(() => {
          setItems((prev) => prev.map((i) => (i.id === id ? { ...i, status: 'done', progress: 100 } : i)))
          files.current.delete(id)
        })
        .catch((e: unknown) => {
          setItems((prev) => prev.map((i) => (i.id === id ? { ...i, status: 'failed', error: e instanceof Error ? e.message : 'Neizdevās' } : i)))
        })
    }
  }, [items])

  const retry = useCallback((ids?: string[]) => setItems((prev) => retryFailed(prev, ids)), [])

  // Finished items leave the queue list once everything settles
  const clearDone = useCallback(() => {
    setItems((prev) => {
      prev.filter((i) => i.status === 'done').forEach((i) => {
        const url = previews.current.get(i.id)
        if (url) URL.revokeObjectURL(url)
        previews.current.delete(i.id)
      })
      return prev.filter((i) => i.status !== 'done')
    })
  }, [])

  return { items, add, retry, clearDone, summary: queueSummary(items), previewOf: (id: string) => previews.current.get(id) }
}
```

```tsx
import { Star, Trash2, Check } from 'lucide-react'

export type GalleryImageRow = { id: string; image_url: string; caption: string | null; display_order: number }

function Tile({ image, isCover, selecting, selected, onToggle, onCover, onDelete, onCaption }: {
  image: GalleryImageRow; isCover: boolean; selecting: boolean; selected: boolean
  onToggle: () => void; onCover: () => void; onDelete: () => void; onCaption: (caption: string) => void
}) {
  const { attributes, listeners, setNodeRef, transform, transition, isDragging } = useSortable({ id: image.id, disabled: selecting })
  const [caption, setCaption] = useState(image.caption ?? '')
  return (
    <li ref={setNodeRef} style={{ transform: CSS.Transform.toString(transform), transition }} className={`grid gap-1.5 ${isDragging ? 'z-10 opacity-70' : ''}`}>
      <div className="relative aspect-square overflow-hidden rounded-md bg-[#EEF2EF]">
        {/* The photo itself is the drag handle; buttons and caption stay clickable */}
        <div {...attributes} {...listeners} className={`h-full w-full touch-none ${selecting ? 'cursor-pointer' : 'cursor-grab'}`} onClick={selecting ? onToggle : undefined}>
          {/* eslint-disable-next-line @next/next/no-img-element */}
          <img src={image.image_url} alt={caption} className="h-full w-full object-cover" draggable={false} loading="lazy" />
        </div>
        {isCover && <span className="absolute left-1.5 top-1.5 inline-flex items-center gap-1 rounded bg-white/95 px-1.5 py-0.5 text-xs font-bold text-club"><Star className="h-3 w-3 fill-current" /> Vāks</span>}
        {selecting ? (
          <span className={`absolute right-1.5 top-1.5 grid h-6 w-6 place-items-center rounded-full border-2 border-white ${selected ? 'bg-teal-700' : 'bg-black/30'}`}>{selected && <Check className="h-4 w-4 text-white" />}</span>
        ) : (
          <div className="absolute right-1.5 top-1.5 flex gap-1">
            {!isCover && <button type="button" onClick={onCover} aria-label="Padarīt par vāku" className="grid h-7 w-7 place-items-center rounded bg-white/95 text-club"><Star className="h-4 w-4" /></button>}
            <button type="button" onClick={onDelete} aria-label="Dzēst foto" className="grid h-7 w-7 place-items-center rounded bg-white/95 text-[#B42318]"><Trash2 className="h-4 w-4" /></button>
          </div>
        )}
      </div>
      <input
        value={caption}
        onChange={(e) => setCaption(e.target.value)}
        onBlur={() => { if (caption !== (image.caption ?? '')) onCaption(caption) }}
        placeholder="Paraksts"
        maxLength={255}
        aria-label="Foto paraksts"
        className="w-full rounded border border-transparent bg-transparent px-1 py-0.5 text-sm hover:border-[#DCE3DF] focus:border-club focus:outline-none"
      />
    </li>
  )
}

export function PhotoGrid({ images, coverUrl, selecting, selected, onToggle, onReorder, onCover, onDelete, onCaption }: {
  images: GalleryImageRow[]
  coverUrl: string | null
  selecting: boolean
  selected: Set<string>
  onToggle: (id: string) => void
  onReorder: (activeId: string, overId: string) => void
  onCover: (image: GalleryImageRow) => void
  onDelete: (image: GalleryImageRow) => void
  onCaption: (image: GalleryImageRow, caption: string) => void
}) {
  const sensors = useSensors(
    useSensor(MouseSensor, { activationConstraint: { distance: 6 } }),
    useSensor(TouchSensor, { activationConstraint: { delay: 250, tolerance: 8 } }),
    useSensor(KeyboardSensor, { coordinateGetter: sortableKeyboardCoordinates })
  )
  const onDragEnd = (e: DragEndEvent) => {
    if (e.over && e.active.id !== e.over.id) onReorder(String(e.active.id), String(e.over.id))
  }
  return (
    <DndContext sensors={sensors} collisionDetection={closestCenter} onDragEnd={onDragEnd}>
      <SortableContext items={images.map((i) => i.id)} strategy={rectSortingStrategy}>
        <ul className="grid grid-cols-2 gap-3 sm:grid-cols-3 lg:grid-cols-4">
          {images.map((image) => (
            <Tile
              key={image.id}
              image={image}
              isCover={image.image_url === coverUrl}
              selecting={selecting}
              selected={selected.has(image.id)}
              onToggle={() => onToggle(image.id)}
              onCover={() => onCover(image)}
              onDelete={() => onDelete(image)}
              onCaption={(c) => onCaption(image, c)}
            />
          ))}
        </ul>
      </SortableContext>
    </DndContext>
  )
}
```

```tsx
'use client'

import { useEffect, useMemo, useRef, useState } from 'react'
import { useRouter } from 'next/navigation'
import { ExternalLink } from 'lucide-react'
import { toast } from 'sonner'
import { AdminFormPage } from '@/components/admin/ui/admin-form-page'
import { Field, inputClass } from '@/components/admin/ui/field'
import { apiSend } from '@/lib/admin-fetch'
import { fieldErrors, galleryUpdateSchema } from '@/lib/admin-schemas'
import { galleryCover, moveId } from '@/lib/gallery'
import { uploadWithProgress } from '@/lib/upload-with-progress'
import { PhotoGrid, type GalleryImageRow } from './photo-grid'
import { UploadPanel } from './upload-panel'
import { useUploadQueue } from './use-upload-queue'

export type GalleryRow = { id: string; title: string; description: string | null; cover_image_url: string | null; created_at: string }

export function GalleryEditor({ gallery, initialImages }: { gallery: GalleryRow; initialImages: GalleryImageRow[] }) {
  const router = useRouter()
  const [images, setImages] = useState(initialImages)
  const [cover, setCover] = useState(gallery.cover_image_url)
  const [meta, setMeta] = useState({ title: gallery.title, description: gallery.description ?? '' })
  const [errors, setErrors] = useState<Record<string, string>>({})
  const [dirty, setDirty] = useState(false)
  const [saving, setSaving] = useState(false)
  const [selecting, setSelecting] = useState(false)
  const [selected, setSelected] = useState<Set<string>>(new Set())
  const nextOrder = useRef(Math.max(0, ...initialImages.map((i) => i.display_order)) + 1)

  const queue = useUploadQueue(async (file, order, onProgress) => {
    const { url } = await uploadWithProgress(file, `galleries/${gallery.id}`, onProgress)
    const row = await apiSend<GalleryImageRow>(`/api/admin/gallery/${gallery.id}/images`, 'POST', { image_url: url, display_order: order })
    setImages((prev) => [...prev, row].sort((a, b) => a.display_order - b.display_order))
  })
  const uploading = queue.summary.active > 0 || queue.summary.waiting > 0

  // Leaving while photos upload would stop them
  useEffect(() => {
    if (!uploading) {
      if (queue.summary.total > 0 && queue.summary.failed === 0) queue.clearDone()
      return
    }
    const warn = (e: BeforeUnloadEvent) => { e.preventDefault(); e.returnValue = '' }
    window.addEventListener('beforeunload', warn)
    return () => window.removeEventListener('beforeunload', warn)
  }, [uploading, queue])

  const shownCover = useMemo(() => galleryCover(cover, images[0]?.image_url), [cover, images])

  const addFiles = (files: File[]) => {
    const start = nextOrder.current
    nextOrder.current += files.length
    queue.add(files, start)
  }

  const reorder = async (activeId: string, overId: string) => {
    const ids = moveId(images.map((i) => i.id), activeId, overId)
    const previous = images
    setImages(ids.map((id, index) => ({ ...images.find((i) => i.id === id)!, display_order: index + 1 })))
    try {
      await apiSend(`/api/admin/gallery/${gallery.id}/order`, 'PUT', { ids })
    } catch (e) {
      setImages(previous)
      toast.error(e instanceof Error ? e.message : 'Neizdevās saglabāt secību')
    }
  }

  const setAsCover = async (image: GalleryImageRow) => {
    try {
      await apiSend(`/api/admin/gallery/${gallery.id}`, 'PATCH', { cover_image_url: image.image_url })
      setCover(image.image_url)
      toast.success('Vāks nomainīts')
    } catch (e) {
      toast.error(e instanceof Error ? e.message : 'Neizdevās')
    }
  }

  const deleteImages = async (targets: GalleryImageRow[]) => {
    const results = await Promise.allSettled(
      targets.map((t) => apiSend<{ cover_cleared: boolean }>(`/api/admin/gallery/${gallery.id}/images/${t.id}`, 'DELETE'))
    )
    const deleted = targets.filter((_, i) => results[i].status === 'fulfilled')
    setImages((prev) => prev.filter((i) => !deleted.some((d) => d.id === i.id)))
    if (deleted.some((d) => d.image_url === cover)) setCover(null)
    if (deleted.length < targets.length) toast.error(`Neizdevās dzēst ${targets.length - deleted.length} foto`)
    else toast.success(deleted.length === 1 ? 'Foto dzēsts' : `Dzēsti ${deleted.length} foto`)
  }

  const saveCaption = async (image: GalleryImageRow, caption: string) => {
    try {
      const row = await apiSend<GalleryImageRow>(`/api/admin/gallery/${gallery.id}/images/${image.id}`, 'PATCH', { caption })
      setImages((prev) => prev.map((i) => (i.id === row.id ? row : i)))
    } catch (e) {
      toast.error(e instanceof Error ? e.message : 'Neizdevās saglabāt parakstu')
    }
  }

  const saveMeta = async () => {
    const result = galleryUpdateSchema.safeParse(meta)
    if (!result.success) { setErrors(fieldErrors(result.error)); return }
    setErrors({})
    setSaving(true)
    try {
      await apiSend(`/api/admin/gallery/${gallery.id}`, 'PATCH', result.data)
      setDirty(false)
      toast.success('Saglabāts')
      router.refresh()
    } catch (e) {
      toast.error(e instanceof Error ? e.message : 'Neizdevās saglabāt')
    } finally {
      setSaving(false)
    }
  }

  const removeGallery = async () => {
    try {
      await apiSend(`/api/admin/gallery/${gallery.id}`, 'DELETE')
      setDirty(false)
      toast.success('Galerija dzēsta')
      router.push('/admin/gallery')
      router.refresh()
    } catch (e) {
      toast.error(e instanceof Error ? e.message : 'Neizdevās dzēst galeriju')
    }
  }

  const toggle = (id: string) => setSelected((prev) => {
    const next = new Set(prev)
    if (next.has(id)) next.delete(id)
    else next.add(id)
    return next
  })

  return (
    <AdminFormPage backHref="/admin/gallery" backLabel="Galerijas" title={meta.title || 'Galerija'} saving={saving} dirty={dirty} onSave={saveMeta} onDelete={removeGallery} deleteTitle="Dzēst visu galeriju un tās foto?">
      <a href={`/gallery/${gallery.id}`} target="_blank" rel="noopener noreferrer" className="inline-flex items-center gap-1.5 justify-self-start text-sm font-semibold text-club">
        Skatīt mājaslapā <ExternalLink className="h-4 w-4" />
      </a>
      <Field label="Nosaukums" htmlFor="title" error={errors.title}>
        <input id="title" value={meta.title} onChange={(e) => { setMeta((m) => ({ ...m, title: e.target.value })); setDirty(true) }} className={inputClass} />
      </Field>
      <Field label="Apraksts" htmlFor="desc">
        <textarea id="desc" rows={2} value={meta.description} onChange={(e) => { setMeta((m) => ({ ...m, description: e.target.value })); setDirty(true) }} className={inputClass} />
      </Field>

      <UploadPanel items={queue.items} summary={queue.summary} onFiles={addFiles} onRetry={queue.retry} previewOf={queue.previewOf} />

      <div className="flex items-center gap-3">
        <h2 className="font-semibold">Foto ({images.length})</h2>
        <span className="flex-1" />
        {selecting && selected.size > 0 && (
          <button type="button" onClick={() => { if (window.confirm(`Dzēst ${selected.size} foto?`)) { deleteImages(images.filter((i) => selected.has(i.id))); setSelected(new Set()); setSelecting(false) } }} className="rounded-md bg-[#B42318] px-3 py-1.5 text-sm font-semibold text-white">
            Dzēst ({selected.size})
          </button>
        )}
        {images.length > 0 && (
          <button type="button" onClick={() => { setSelecting((s) => !s); setSelected(new Set()) }} className="rounded-md border border-[#DCE3DF] px-3 py-1.5 text-sm font-semibold">
            {selecting ? 'Atcelt' : 'Izvēlēties'}
          </button>
        )}
      </div>
      {images.length === 0 ? (
        <p className="rounded-lg border border-dashed border-[#DCE3DF] p-6 text-center text-[#5B6863]">Galerijā vēl nav foto.</p>
      ) : (
        <PhotoGrid
          images={images}
          coverUrl={shownCover}
          selecting={selecting}
          selected={selected}
          onToggle={toggle}
          onReorder={reorder}
          onCover={setAsCover}
          onDelete={(image) => { if (window.confirm('Dzēst šo foto?')) deleteImages([image]) }}
          onCaption={saveCaption}
        />
      )}
    </AdminFormPage>
  )
}
```
